// Robust Excel Upload Parser with Diagnostic Alerts function handleExcelUpload(event) { const fileInput = event.target; const file = fileInput.files[0]; if (!file) { alert("No file was selected."); return; } // Check if SheetJS library is loaded properly if (typeof XLSX === 'undefined') { alert("Error: Excel parser library (SheetJS) failed to load. Check your internet connection or ad-blocker settings."); fileInput.value = ''; return; } document.getElementById('fileUploadLabel').innerText = `📄 ${file.name}`; const reader = new FileReader(); reader.onerror = function(err) { alert("Failed to read file from disk: " + err); fileInput.value = ''; }; reader.onload = function(e) { try { const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); if (!workbook.SheetNames || workbook.SheetNames.length === 0) { alert("The Excel file contains no worksheets."); return; } const firstSheetName = workbook.SheetNames[0]; const worksheet = workbook.Sheets[firstSheetName]; // Read sheet as 2D Array const rows = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: "" }); if (!rows || rows.length === 0) { alert("The selected worksheet is completely empty."); return; } let importedCount = 0; let headerIdx = -1; let colMap = { slNo: -1, name: -1, parent: -1, mobile: -1, aadhaar: -1, gender: -1 }; // Scan first 15 rows to find header row dynamically for (let i = 0; i < Math.min(rows.length, 15); i++) { const rowCells = rows[i].map(cell => String(cell).trim().toLowerCase().replace(/[^a-z0-9]/g, '')); const slIdx = rowCells.findIndex(c => c === 'slno' || c === 'sl' || c === 'serialno' || c === 'serial' || c === 'sno'); const nameIdx = rowCells.findIndex(c => c === 'name' || c === 'votername' || c === 'fullname'); if (slIdx !== -1 || nameIdx !== -1) { headerIdx = i; colMap.slNo = slIdx; colMap.name = nameIdx; colMap.parent = rowCells.findIndex(c => c.includes('relation') || c.includes('parent') || c.includes('husband') || c.includes('father')); colMap.mobile = rowCells.findIndex(c => c.includes('mobile') || c.includes('phone') || c.includes('cell')); colMap.aadhaar = rowCells.findIndex(c => c.includes('aadhaar') || c.includes('aadhar')); colMap.gender = rowCells.findIndex(c => c.includes('gender') || c.includes('sex')); break; } } const startRow = headerIdx !== -1 ? headerIdx + 1 : 0; for (let i = startRow; i < rows.length; i++) { const row = rows[i]; if (!row || row.length === 0) continue; let slNo = colMap.slNo !== -1 ? String(row[colMap.slNo] || '').trim() : String(row[0] || '').trim(); let name = colMap.name !== -1 ? String(row[colMap.name] || '—').trim() : String(row[1] || '—').trim(); let parent = colMap.parent !== -1 ? String(row[colMap.parent] || '—').trim() : String(row[2] || '—').trim(); let mobile = colMap.mobile !== -1 ? String(row[colMap.mobile] || '—').trim() : String(row[3] || '—').trim(); let genderVal = colMap.gender !== -1 ? row[colMap.gender] : row[4]; let aadhaar = colMap.aadhaar !== -1 ? String(row[colMap.aadhaar] || '—').trim() : String(row[5] || '—').trim(); if (!slNo || slNo.toLowerCase().includes('sl') || name.toLowerCase().includes('name')) continue; if (!(window.voterStore || []).some(v => String(v.slNo) === String(slNo))) { window.voterStore.push({ slNo, name, parent, mobile, aadhaar, gender: parseGender(genderVal), status: 'active' }); importedCount++; } } if (typeof window.syncDatabase === 'function') { window.syncDatabase(); } alert(`Import completed! ${importedCount} new voters were added.`); } catch (err) { console.error(err); alert("Error parsing file: " + err.message); } finally { fileInput.value = ''; } }; reader.readAsArrayBuffer(file); }
